```python
words = ['delta', 'pine', 'ivory']
out = ""
for index in range(len(words)):
    piece = words[index]
    if index % 3 == 0:
        piece = piece.upper()
    if out == "":
        out = piece
    else:
        out = out + '/' + piece
print(out)
```
